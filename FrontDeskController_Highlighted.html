156    private void initializeNextGuestConfirmationNumber() {
157        int highest = 10000000;
158        ListInterface<Guest> guests = guestTree.inOrderTraversal();
159        for (int i = 0; i < guests.getNumberOfEntries(); i++) {
160            String confirmation = guests.get(i).getConfirmationNumber();
161            if (confirmation != null && confirmation.matches("\\d{8}"))
162                highest = Math.max(highest, Integer.parseInt(confirmation));
163        }
164        nextGuestConfirmationNumber = highest + 1;
165    }

167    private void syncRoomTree() {
168        if (sharedRoomList == null || syncedRoomCount == sharedRoomList.getNumberOfEntries())
169            return;
170​
171        // Room objects are shared references, so status/price/type changes are
172        // already visible inside the tree. Rebuild only when the list structure
173        // changes. Keep the insertion-order shape so the diagnostic menu can
174        // demonstrate the visible effect of an explicit BST rebalance.
175        roomTree.clear();
176        for (int i = 0; i < sharedRoomList.getNumberOfEntries(); i++)
177            roomTree.add(sharedRoomList.get(i));
178        syncedRoomCount = sharedRoomList.getNumberOfEntries();
179    }

258     * Registers a guest while keeping confirmation-number generation inside the
259     * controller. Return status: 1 success, -1 invalid data, -3 duplicate
260     * IC/passport, -4 no 8-digit confirmation number remains available.
261     */
262    public GuestRegistrationResult registerGuestWithGeneratedConfirmation(
263            String name, String icNo, String phoneNumber) {
264        if (isMissingRequired(name) || isMissingRequired(icNo) || !isValidOptionalPhone(phoneNumber))
265            return new GuestRegistrationResult(-1, null);
266        if (searchGuestByIC(icNo) != null)
267            return new GuestRegistrationResult(-3, null);
268​
269        while (nextGuestConfirmationNumber <= 99999999
270                && searchGuestByConfirmationNumber(String.format("%08d", nextGuestConfirmationNumber)) != null)
271            nextGuestConfirmationNumber++;
272        if (nextGuestConfirmationNumber > 99999999)
273            return new GuestRegistrationResult(-4, null);
274​
275        String confirmation = String.format("%08d", nextGuestConfirmationNumber++);
276        Guest guest = new Guest(name, icNo, phoneNumber, confirmation, "Standard", 0);
277        return addValidatedGuest(guest)
278                ? new GuestRegistrationResult(1, guest)
279                : new GuestRegistrationResult(-1, null);
280    }

307    public int updateGuestProfile(String confirmationNumber, String name, String icNo,
308            String phone, String gender, String nationality, String email, String specialRequest) {
309        Guest guest = searchGuestByConfirmationNumber(confirmationNumber);
310        if (guest == null)
311            return -1;
312        if (isMissingRequired(name) || isMissingRequired(icNo))
313            return -2;
314​
315        boolean identityUnchanged = ControllerDataSupport.normalizeIdentity(guest.getIcNo())
316                .equals(ControllerDataSupport.normalizeIdentity(icNo));
317        Guest sameIdentity = searchGuestByIC(icNo);
318        if (sameIdentity != null && sameIdentity != guest && !identityUnchanged)
319            return -3;
320        if (!isBlank(email) && !"N/A".equalsIgnoreCase(email.trim())
321                && !email.trim().matches("^[^@\\s]+@[^@\\s]+\\.[^@\\s]+$"))
322            return -4;
323        if (!isValidOptionalPhone(phone))
324            return -5;
325​
326        if (!identityUnchanged && !loyaltyController.migrateMemberIdentity(guest, icNo.trim()))
327            return -6;
328​
329        guest.setGuestName(name.trim());
330        guest.setIcNo(icNo.trim());
331        guest.setPhoneNumber(normalizeOptional(phone));
332        guest.setGender(normalizeOptional(gender));
333        guest.setNationality(isBlank(nationality) ? "N/A" : nationality.trim());
334        guest.setEmail(normalizeOptional(email));
335        guest.setSpecialRequest(isBlank(specialRequest) || "N/A".equalsIgnoreCase(specialRequest.trim())
336                ? null
337                : specialRequest.trim());
338​
339        // Booking stores display snapshots, so keep them aligned with the master guest.
340        for (int i = 0; i < sharedBookingList.getNumberOfEntries(); i++) {
341            Booking booking = sharedBookingList.get(i);
342            if (booking != null && confirmationNumber.equalsIgnoreCase(booking.getGuestConfirmationNumber())) {
343                booking.setGuestName(guest.getGuestName());
344                booking.setSpecialRequest(guest.getSpecialRequest());
345            }
346        }
347        return 1;
348    }

506    /**
507     * Searches rooms by date range, room type and budget, then sorts by price.
508     * This uses the shared booking schedule rather than only today's physical
509     * status.
510     */
511    public ListInterface<Room> searchAvailableRooms(String checkInDate, int numberOfNights,
512            String roomTypeFilter, double maxPrice, boolean sortAscending) {
513        ListInterface<Room> result = new MyArrayList<>();
514        if (validateStayPeriod(checkInDate, numberOfNights) != 1 || maxPrice < 0)
515            return result;
516        syncRoomTree();
517        LocalDate start = LocalDate.parse(checkInDate.trim());
518        LocalDate end = start.plusDays(numberOfNights);
519        ListInterface<Room> rooms = roomTree.inOrderTraversal();
520        for (int i = 0; i < rooms.getNumberOfEntries(); i++) {
521            Room room = rooms.get(i);
522            boolean typeMatches = isBlank(roomTypeFilter) || "ALL".equalsIgnoreCase(roomTypeFilter.trim())
523                    || room.getRoomType().equalsIgnoreCase(roomTypeFilter.trim());
524            boolean budgetMatches = maxPrice == 0 || room.getPrice() <= maxPrice;
525            if (!typeMatches || !budgetMatches || !isRoomSellableForPeriod(room, start, end))
526                continue;
527            result.add(room);
528        }
529        result.sort((left, right) -> sortAscending
530                ? Double.compare(left.getPrice(), right.getPrice())
531                : Double.compare(right.getPrice(), left.getPrice()));
532        return result;
533    }

605    public int processCheckIn(String confirmationNumber, String roomNumber, double baseRoomPrice) {
606        syncRoomTree();
607        Guest guest = searchGuestByConfirmationNumber(confirmationNumber);
608        if (guest == null)
609            return -1;
610​
611        // Guest.bookingStatus is the single source of truth for check-in state.
612        if (guest.isCheckedIn()) {
613            return -4;
614        }
615​
616        if ("CheckedOut".equalsIgnoreCase(guest.getBookingStatus())) {
617            return -6; // Guest already checked out for this stay
618        }
619​
620        if ("Cancelled".equalsIgnoreCase(guest.getBookingStatus())) {
621            return -7; // Booking was cancelled
622        }
623​
624        if ("NoShow".equalsIgnoreCase(guest.getBookingStatus()))
625            return -8;
626​
627        if (!guest.isReserved())
628            return -12;
629​
630        LocalDate today = LocalDate.now();
631        Booking booking = findBookingByConfirmation(confirmationNumber);
632        if (booking == null)
633            return -13;
634        if ("NoShow".equalsIgnoreCase(booking.getBookingStatus()))
635            return -8;
636        if (!"Confirmed".equalsIgnoreCase(booking.getBookingStatus()))
637            return -13;
638        // Validate the complete scheduled stay before making any state change. In

689                }
690            }

693        room.setRoomStatus("Occupied");

697        guest.setAssignedRoomNumber(roomNumber.trim());
698        guest.setEffectiveRoomRate(baseRoomPrice > 0 ? baseRoomPrice : room.getPrice());
699        guest.setRoomType(room.getRoomType());
700        if (booking != null) {
701            booking.recordCheckIn();
702            booking.setRoomNumber(room.getRoomNumber());
703            booking.setRoomType(room.getRoomType());
704        }
705        return 1;
706    }
